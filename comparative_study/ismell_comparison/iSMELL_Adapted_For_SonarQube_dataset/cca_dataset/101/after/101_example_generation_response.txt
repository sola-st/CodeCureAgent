// Noncompliant code example
public class BitOperations {
    public static int sillyBitOps(int value) {
        int result1 = value & -1;  // silly: result1 == value
        int result2 = value ^ 0;   // silly: result2 == value
        int result3 = value | 0;   // silly: result3 == value
        return result1 + result2 + result3;
    }
}

// Compliant solution
public class BitOperations {
    public static int correctBitOps(int value) {
        // Remove silly bit operations; just use the original value
        int result1 = value;
        int result2 = value;
        int result3 = value;
        return result1 + result2 + result3;
    }
}
