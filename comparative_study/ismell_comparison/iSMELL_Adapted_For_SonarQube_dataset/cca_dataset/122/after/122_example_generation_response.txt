// Compliant solution
import java.util.Map;

public class FruitColor {
    private static final Map<String, String> FRUIT_COLOR_MAP = Map.ofEntries(
        Map.entry("apple", "red"),
        Map.entry("strawberry", "red"),
        Map.entry("banana", "yellow"),
        Map.entry("lemon", "yellow"),
        Map.entry("lime", "green"),
        Map.entry("kiwi", "green"),
        Map.entry("blueberry", "purple"),
        Map.entry("grape", "purple"),
        Map.entry("orange", "orange"),
        Map.entry("pear", "greenish yellow"),
        Map.entry("watermelon", "green and red"),
        Map.entry("cherry", "red"),
        Map.entry("peach", "pinkish-orange"),
        Map.entry("plum", "purple")
        // Add more mappings as needed
    );

    public String getColor(String fruit) {
        return FRUIT_COLOR_MAP.getOrDefault(fruit, "unknown");
    }
}
