```java
// Noncompliant code example
public class GarbageCollectionTrigger {
    public void forceGarbageCollection() {
        // Explicit call to the garbage collector — not recommended
        System.gc();
        Runtime.getRuntime().gc();
        System.runFinalization();
    }
}

// Compliant solution
public class GarbageCollectionTrigger {
    public void doWork() {
        // Let the JVM handle garbage collection and finalization automatically
        // No explicit calls to System.gc(), Runtime.gc(), or System.runFinalization()
        // Perform normal application logic here
    }
}
```
