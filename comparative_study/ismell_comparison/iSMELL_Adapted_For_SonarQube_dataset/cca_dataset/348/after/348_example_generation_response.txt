// Compliant solution
public class OrderProcessor {
    public void processOrder(Order order) {
        if (order == null || !order.isValid() || !order.isPaid() || !order.isInStock()
            || !order.isShipmentAddressValid() || !order.isUserAuthorized() || order.getItems().isEmpty()) {
            return;
        }
        processItems(order);
    }

    private void processItems(Order order) {
        for (Item item : order.getItems()) {
            if (item.isAvailable() && item.isNotDamaged()) {
                // process each item
            }
        }
    }
}
