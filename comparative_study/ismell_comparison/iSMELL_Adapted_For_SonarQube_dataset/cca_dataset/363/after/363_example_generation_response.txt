```java
// Compliant solution
public class OrderProcessor {
  public String processOrder(int orderType, boolean isUrgent, boolean hasDiscount, boolean isInternational, boolean isBackOrder) {
    if (orderType == 1) {
      return processStandardOrder(isUrgent, hasDiscount, isInternational, isBackOrder);
    } else if (orderType == 2) {
      return processSpecialOrder();
    } else {
      return "Unknown order type";
    }
  }

  private String processStandardOrder(boolean isUrgent, boolean hasDiscount, boolean isInternational, boolean isBackOrder) {
    if (isUrgent) {
      return processUrgentOrder(hasDiscount, isInternational, isBackOrder);
    } else if (hasDiscount) {
      return "Process standard order with discount";
    } else {
      return "Process standard order";
    }
  }

  private String processUrgentOrder(boolean hasDiscount, boolean isInternational, boolean isBackOrder) {
    if (hasDiscount) {
      if (isInternational) {
        if (isBackOrder) {
          return "Process urgent international backorder with discount";
        } else {
          return "Process urgent international order with discount";
        }
      } else {
        return "Process urgent domestic order with discount";
      }
    } else if (isInternational) {
      return "Process urgent international order without discount";
    } else {
      return "Process urgent domestic order without discount";
    }
  }

  private String processSpecialOrder() {
    return "Process special order";
  }
}
```
