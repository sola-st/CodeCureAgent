```java
// Compliant solution
public class OrderProcessor {
    public void processOrder(Order order) {
        if (order == null) {
            System.out.println("Order is null.");
            return;
        }
        if (!order.isPaid()) {
            System.out.println("Order not paid.");
            return;
        }
        if (!order.hasItems()) {
            System.out.println("Order has no items.");
            return;
        }

        for (Item item : order.getItems()) {
            processItem(item);
        }
    }

    private void processItem(Item item) {
        if (!item.isInStock()) {
            System.out.println("Out of stock item: " + item.getName());
            return;
        }
        if (item.getQuantity() <= 0) {
            System.out.println("Invalid quantity for item: " + item.getName());
            return;
        }
        if (item.isExpired()) {
            System.out.println("Expired item: " + item.getName());
            return;
        }
        if (item.getPrice() <= 0) {
            System.out.println("Invalid price for item: " + item.getName());
            return;
        }
        // Process valid item
        System.out.println("Processing item: " + item.getName());
    }
}
```
