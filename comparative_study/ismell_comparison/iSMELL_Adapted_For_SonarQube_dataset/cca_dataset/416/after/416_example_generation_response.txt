```java
// Compliant solution
public class OrderProcessor {

    public String processOrder(int orderType, boolean isUrgent, boolean hasDiscount, boolean isInternational) {
        StringBuilder sb = new StringBuilder("Process ");
        
        if (isUrgent) {
            sb.append("urgent ");
        }
        if (isInternational) {
            sb.append("international ");
        }
        if (hasDiscount) {
            sb.append("discounted ");
        }
        sb.append("type ").append(orderType).append(" order");
        
        if (orderType != 1 && orderType != 2) {
            return "Unknown order type";
        }
        
        return sb.toString();
    }
}
```
