```java
// Compliant solution
public class OrderProcessor {
    enum CustomerType { VIP, REGULAR }
    enum OrderType { NORMAL, SPECIAL }
    public void processOrder(int orderTypeVal, int quantity, boolean isExpress, boolean isGift, int customerTypeVal) {
        OrderType orderType = (orderTypeVal == 1) ? OrderType.NORMAL : OrderType.SPECIAL;
        CustomerType customerType = (customerTypeVal == 1) ? CustomerType.VIP : CustomerType.REGULAR;

        if (orderType == OrderType.SPECIAL) {
            System.out.println("Process special order type 2");
            return;
        }

        if (quantity > 10) {
            processBulkOrder(isExpress, isGift, customerType);
        } else {
            processSmallOrder(isExpress, isGift);
        }
    }

    private void processBulkOrder(boolean isExpress, boolean isGift, CustomerType customerType) {
        if (isExpress && isGift && customerType == CustomerType.VIP) {
            System.out.println("Process VIP express gift bulk order");
        } else if (isExpress && isGift) {
            System.out.println("Process express gift bulk order");
        } else if (isExpress && customerType == CustomerType.VIP) {
            System.out.println("Process VIP express bulk order");
        } else if (isExpress) {
            System.out.println("Process express bulk order");
        } else if (isGift && customerType == CustomerType.VIP) {
            System.out.println("Process VIP gift bulk order");
        } else if (isGift) {
            System.out.println("Process gift bulk order");
        } else if (customerType == CustomerType.VIP) {
            System.out.println("Process VIP bulk order");
        } else {
            System.out.println("Process bulk order");
        }
    }

    private void processSmallOrder(boolean isExpress, boolean isGift) {
        if (isExpress && isGift) {
            System.out.println("Process express gift small order");
        } else if (isExpress) {
            System.out.println("Process express small order");
        } else if (isGift) {
            System.out.println("Process gift small order");
        } else {
            System.out.println("Process small order");
        }
    }
}
```
