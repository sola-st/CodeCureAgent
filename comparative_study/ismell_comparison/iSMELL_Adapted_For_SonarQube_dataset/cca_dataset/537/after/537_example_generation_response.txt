```java
// Compliant solution
public class OrderProcessor {

    public String processOrder(int orderType, int paymentType, boolean isPriority, boolean isInternational, boolean hasDiscount, int shippingMethod) {
        if (orderType == 1) {
            return handleOrderTypeOne(paymentType, isPriority, isInternational, hasDiscount, shippingMethod);
        } else if (orderType == 2) {
            return handleOrderTypeTwo();
        } else {
            return "Unknown order type";
        }
    }

    private String handleOrderTypeOne(int paymentType, boolean isPriority, boolean isInternational, boolean hasDiscount, int shippingMethod) {
        if (paymentType == 1) {
            return handleCreditCardOrder(isPriority, isInternational, hasDiscount);
        } else if (paymentType == 2) {
            return handlePaypalOrder(shippingMethod);
        } else {
            return "Unsupported payment method";
        }
    }

    private String handleCreditCardOrder(boolean isPriority, boolean isInternational, boolean hasDiscount) {
        if (isPriority) {
            if (isInternational) {
                if (hasDiscount) {
                    return "Process priority international order with discount via credit card";
                }
                return "Process priority international order via credit card";
            }
            return "Process priority domestic order via credit card";
        }
        return "Process regular order via credit card";
    }

    private String handlePaypalOrder(int shippingMethod) {
        if (shippingMethod == 1) {
            return "Process order with express shipping and PayPal";
        }
        return "Process order with standard shipping and PayPal";
    }

    private String handleOrderTypeTwo() {
        // Implementation for orderType 2
        return "Process different order type";
    }
}
```
