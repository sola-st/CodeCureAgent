```java
// Compliant solution
public class OrderProcessor {

    public String processOrder(int orderType, boolean isPriority, boolean isInternational, boolean isGift, int customerLevel) {
        if (orderType == 1) {
            return processTypeOneOrder(isPriority, isInternational, isGift, customerLevel);
        }
        if (orderType == 2) {
            return processTypeTwoOrder(isGift);
        }
        return "Unknown order type";
    }

    private String processTypeOneOrder(boolean isPriority, boolean isInternational, boolean isGift, int customerLevel) {
        if (!isPriority) {
            return "Process normal order";
        }
        if (!isInternational) {
            return "Process with priority flow";
        }
        if (!isGift) {
            return "Process with international priority flow";
        }
        if (customerLevel > 5) {
            return "Process with international priority gift flow for premium customer";
        }
        return "Process with international priority gift flow";
    }

    private String processTypeTwoOrder(boolean isGift) {
        if (isGift) {
            return "Process gift order type 2";
        }
        return "Process order type 2";
    }
}
```
