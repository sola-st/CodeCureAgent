    FeedScopedId route;

    FeedScopedId trip;

    Set<String> zones;

    String startZone;

    String endZone;

    long startTime;

    long endTime;

        builder.append("Ride");
        if (startZone != null) {
            builder.append("(from zone ");
            builder.append(startZone);
        }
        if (endZone != null) {
            builder.append(" to zone ");
            builder.append(endZone);
        }
        builder.append(" on route ");
        builder.append(route);
        if (zones.size() > 0) {

            boolean first = true;
            for (String zone : zones) {
                if (first) {
                    first = false;
                } else {

                }
                builder.append(zone);
            }
        }
        builder.append(" at ");
        builder.append(startTime);
        if (classifier != null) {
            builder.append(", classified by ");
            builder.append(classifier.toString());
        }
        builder.append(")");
