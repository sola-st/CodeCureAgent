```java
// Compliant solution
public class OrderProcessor {
    public String processOrder(int orderType, boolean isPriority, int quantity, boolean isGift, String paymentMethod) {
        if (orderType == 1) {
            return processType1Order(isPriority, quantity, isGift, paymentMethod);
        } else if (orderType == 2) {
            return processType2Order(isGift, paymentMethod);
        } else {
            return processOtherOrder(paymentMethod);
        }
    }

    private String processType1Order(boolean isPriority, int quantity, boolean isGift, String paymentMethod) {
        if (isPriority) {
            if (quantity > 10) {
                return isGift ? "Process large priority gift order type 1 with " + paymentMethod
                        : "Process large priority order type 1 with " + paymentMethod;
            } else {
                return "Process small priority order type 1 with " + paymentMethod;
            }
        } else {
            if (quantity > 10) {
                return "Process large regular order type 1 with " + paymentMethod;
            } else {
                return "Process small regular order type 1 with " + paymentMethod;
            }
        }
    }

    private String processType2Order(boolean isGift, String paymentMethod) {
        return isGift ? "Process gift order type 2 with " + paymentMethod
                : "Process regular order type 2 with " + paymentMethod;
    }

    private String processOtherOrder(String paymentMethod) {
        switch (paymentMethod) {
            case "card":
                return "Process order with card payment";
            case "paypal":
                return "Process order with PayPal payment";
            default:
                return "Process order with other payment method";
        }
    }
}
```
