```java
// Compliant solution
public class OrderProcessor {

    public String processOrder(int customerType, boolean hasDiscount, int orderPriority, boolean isHoliday, boolean isExpress, boolean isInternational) {
        if (customerType == 1) {
            return processVipOrder(hasDiscount, orderPriority, isHoliday, isExpress, isInternational);
        } else if (customerType == 2) {
            return processRegularOrder(hasDiscount);
        } else {
            return "Process guest order";
        }
    }

    private String processVipOrder(boolean hasDiscount, int orderPriority, boolean isHoliday, boolean isExpress, boolean isInternational) {
        if (hasDiscount) {
            if (orderPriority > 5) {
                if (isHoliday) {
                    return "Process with discount for VIP customer on holiday with high priority";
                } else {
                    return "Process with discount for VIP customer with high priority";
                }
            }
            return "Process with discount for VIP customer";
        } else if (orderPriority > 5) {
            if (isExpress) {
                if (isInternational) {
                    return "Process express international VIP order with high priority";
                } else {
                    return "Process express VIP order with high priority";
                }
            }
            return "Process VIP order with high priority";
        }
        return "Process VIP order normally";
    }

    private String processRegularOrder(boolean hasDiscount) {
        if (hasDiscount) {
            return "Process discount for regular customer";
        }
        return "Process regular order";
    }
}
```
