```java
// Noncompliant code example
public class OrderProcessor {
    public void processOrder(Order order) {
        if (order.isValid()) {
            if (order.getItems() != null) {
                for (Item item : order.getItems()) {
                    if (item.isAvailable()) {
                        if (item.getQuantity() > 0) {
                            if (order.isExpressDelivery()) {
                                if (order.getCustomer().hasValidAddress()) {
                                    System.out.println("Processing express order item: " + item.getName());
                                } else {
                                    System.out.println("Invalid address for express delivery");
                                }
                            } else {
                                System.out.println("Processing standard order item: " + item.getName());
                            }
                        }
                    }
                }
            }
        } else {
            System.out.println("Invalid order");
        }
    }
}
```

```java
// Compliant solution
public class OrderProcessor {
    public void processOrder(Order order) {
        if (!order.isValid()) {
            System.out.println("Invalid order");
            return;
        }
        if (order.getItems() == null) {
            return;
        }
        for (Item item : order.getItems()) {
            processItem(order, item);
        }
    }

    private void processItem(Order order, Item item) {
        if (!item.isAvailable() || item.getQuantity() <= 0) {
            return;
        }
        if (order.isExpressDelivery()) {
            processExpressItem(order, item);
        } else {
            System.out.println("Processing standard order item: " + item.getName());
        }
    }

    private void processExpressItem(Order order, Item item) {
        if (!order.getCustomer().hasValidAddress()) {
            System.out.println("Invalid address for express delivery");
            return;
        }
        System.out.println("Processing express order item: " + item.getName());
    }
}
```
