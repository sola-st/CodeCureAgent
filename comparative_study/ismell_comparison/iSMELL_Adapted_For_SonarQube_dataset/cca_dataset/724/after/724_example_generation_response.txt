```java
// Compliant solution - reduced cognitive complexity by flattening and decomposing
public class OrderProcessor {

    public String processOrder(int orderType, boolean isPriority, int customerType, boolean isVerified) {
        if (orderType == 1) {
            return processType1Order(isPriority, customerType, isVerified);
        } else if (orderType == 2) {
            return processType2Order(isPriority, customerType, isVerified);
        } else {
            return "Unsupported order type";
        }
    }

    private String processType1Order(boolean isPriority, int customerType, boolean isVerified) {
        if (isPriority) {
            return processPriorityOrder(customerType, isVerified, "priority");
        } else {
            return processPriorityOrder(customerType, isVerified, "normal priority");
        }
    }

    private String processType2Order(boolean isPriority, int customerType, boolean isVerified) {
        // simplified example: just for illustration
        return "Process order type 2 with simplified logic";
    }

    private String processPriorityOrder(int customerType, boolean isVerified, String priorityLabel) {
        if (customerType == 1) {
            if (isVerified) {
                return "Process " + priorityLabel + " verified customer order type 1";
            } else {
                return "Process " + priorityLabel + " unverified customer order type 1";
            }
        } else {
            if (isVerified) {
                return "Process " + priorityLabel + " verified non-customer order type 1";
            } else {
                return "Process " + priorityLabel + " unverified non-customer order type 1";
            }
        }
    }
}
```
