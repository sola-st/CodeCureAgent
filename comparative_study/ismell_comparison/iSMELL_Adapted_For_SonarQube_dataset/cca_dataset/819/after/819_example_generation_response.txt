```java
// Compliant solution
public class OrderProcessor {

    public String processOrder(int orderType, int paymentMethod, boolean isExpress, boolean isGift, int customerType) {
        if (orderType == 1) {
            return processType1Order(paymentMethod, isExpress, isGift);
        } else if (orderType == 2) {
            return processType2Order(customerType);
        }
        return "Unknown order";
    }

    private String processType1Order(int paymentMethod, boolean isExpress, boolean isGift) {
        String delivery = isExpress ? "Express" : "Standard";
        String gift = isGift ? "gift " : "";
        String payment = paymentMethod == 1 ? "cash" : paymentMethod == 2 ? "card" : "unknown payment";
        return String.format("%s %sorder with %s payment for type 1", delivery, gift, payment);
    }

    private String processType2Order(int customerType) {
        if (customerType == 1) {
            return "Order type 2 for regular customer";
        } else {
            return "Order type 2 for premium customer";
        }
    }
}
```
