// Compliant solution
public class OrderProcessor {
    public void processOrder(Order order) {
        if (order == null || !order.isValid()) {
            return;
        }
        for (Item item : order.getItems()) {
            processItem(order, item);
        }
    }

    private void processItem(Order order, Item item) {
        if (item.isInStock() && item.getQuantity() > 0 && !item.isExpired()) {
            if (order.isPriority()) {
                // process priority item
            } else if (order.isInternational()) {
                // process international item
            } else {
                // process regular item
            }
        }
    }
}
