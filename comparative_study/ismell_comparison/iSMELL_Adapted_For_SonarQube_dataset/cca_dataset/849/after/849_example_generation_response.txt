```java
// Noncompliant code example
public class OrderProcessor {
    public void processOrder(Order order) {
        if (order != null) {
            if (order.isValid()) {
                for (Item item : order.getItems()) {
                    if (item.isInStock()) {
                        if (item.getQuantity() > 0) {
                            if (item.getPrice() > 0) {
                                System.out.println("Processing item: " + item.getName());
                            } else {
                                System.out.println("Invalid price for item: " + item.getName());
                            }
                        } else {
                            System.out.println("Invalid quantity for item: " + item.getName());
                        }
                    } else {
                        System.out.println("Item out of stock: " + item.getName());
                    }
                }
            } else {
                System.out.println("Order is not valid");
            }
        } else {
            System.out.println("Order is null");
        }
    }
}
```

```java
// Compliant solution
public class OrderProcessor {
    public void processOrder(Order order) {
        if (order == null) {
            System.out.println("Order is null");
            return;
        }
        if (!order.isValid()) {
            System.out.println("Order is not valid");
            return;
        }
        for (Item item : order.getItems()) {
            processItem(item);
        }
    }

    private void processItem(Item item) {
        if (!item.isInStock()) {
            System.out.println("Item out of stock: " + item.getName());
            return;
        }
        if (item.getQuantity() <= 0) {
            System.out.println("Invalid quantity for item: " + item.getName());
            return;
        }
        if (item.getPrice() <= 0) {
            System.out.println("Invalid price for item: " + item.getName());
            return;
        }
        System.out.println("Processing item: " + item.getName());
    }
}
```
