```java
// Compliant solution
public class OrderProcessor {
    public String processOrder(int orderType, boolean isUrgent, boolean isInternational, int paymentMethod, int deliveryOption) {
        if (orderType == 1) {
            return processStandardOrder(isUrgent, isInternational, paymentMethod, deliveryOption);
        } else if (orderType == 2) {
            return "Process subscription order";
        } else {
            return "Unknown order type";
        }
    }

    private String processStandardOrder(boolean isUrgent, boolean isInternational, int paymentMethod, int deliveryOption) {
        String urgency = isUrgent ? "urgent" : "normal";
        String location = isInternational ? "international" : "domestic";
        String payment = paymentMethod == 1 ? "credit card" : "paypal";
        String delivery = deliveryOption == 1 ? "express delivery" : "standard delivery";

        if (isUrgent && isInternational) {
            return String.format("Process %s %s order with %s and %s", urgency, location, payment, delivery);
        } else if (isUrgent) {
            return String.format("Process %s %s order with %s", urgency, location, payment);
        } else if (isInternational) {
            return "Process normal international order";
        } else {
            return "Process normal domestic order";
        }
    }
}
```
