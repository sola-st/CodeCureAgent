```java
// Compliant solution
public class OrderProcessor {

    public String processOrder(int orderType, boolean isPriority, boolean isInternational, int paymentMethod) {
        if (orderType == 1) {
            return processDomesticOrder(isPriority, paymentMethod);
        } 
        if (orderType == 2) {
            if (isInternational) {
                return processInternationalOrder(isPriority, paymentMethod);
            } else {
                return "Invalid international order";
            }
        }
        return "Unknown order type";
    }

    private String processDomesticOrder(boolean isPriority, int paymentMethod) {
        String priority = isPriority ? "priority" : "standard";
        String payment = paymentMethodToString(paymentMethod);
        return "Process domestic " + priority + " " + payment + " order";
    }

    private String processInternationalOrder(boolean isPriority, int paymentMethod) {
        String priority = isPriority ? "priority" : "standard";
        String payment = paymentMethodToString(paymentMethod);
        return "Process international " + priority + " " + payment + " order";
    }

    private String paymentMethodToString(int paymentMethod) {
        switch (paymentMethod) {
            case 1:
                return "cash";
            case 2:
                return "card";
            default:
                return "other payment";
        }
    }
}
```
