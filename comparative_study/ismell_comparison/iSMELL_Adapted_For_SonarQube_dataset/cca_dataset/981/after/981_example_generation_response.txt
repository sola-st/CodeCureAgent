```java
// Compliant solution
public class OrderProcessor {
    public String processOrder(int orderType, boolean isUrgent, boolean isInternational, boolean hasDiscount, boolean isGift) {
        if (orderType != 1) {
            return "Unknown order type";
        }
        StringBuilder result = new StringBuilder("Process ");
        if (isUrgent) {
            result.append("urgent ");
        }
        result.append(isInternational ? "international " : "domestic ");
        if (hasDiscount) {
            result.append("discounted ");
        }
        if (isGift) {
            result.append("gift ");
        }
        result.append("order");
        return result.toString();
    }
}
```
