// Compliant solution
public class OrderProcessor {

    public void processOrder(String orderType, boolean isMember, boolean isExpress, int paymentMethod, int discountCode) {
        if ("online".equals(orderType)) {
            processOnlineOrder(isMember, isExpress, paymentMethod, discountCode);
        } else if ("store".equals(orderType)) {
            processStoreOrder(isExpress, paymentMethod);
        } else {
            processOtherOrder();
        }
    }
    
    private void processOnlineOrder(boolean isMember, boolean isExpress, int paymentMethod, int discountCode) {
        if (isMember) {
            processMemberOnlineOrder(isExpress, paymentMethod);
        } else {
            processGuestOnlineOrder(discountCode);
        }
    }
    
    private void processMemberOnlineOrder(boolean isExpress, int paymentMethod) {
        if (isExpress) {
            processExpressPayment(paymentMethod, "online");

            processNormalPayment(paymentMethod, "online");

    }
