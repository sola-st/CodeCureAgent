```java
// Compliant solution
public class OrderProcessor {

    public void processOrder(int orderType, boolean isPriority, boolean isInternational, boolean isGift, boolean requiresSignature) {
        if (orderType == 1) {
            processType1Order(isPriority, isInternational, isGift, requiresSignature);
        } else if (orderType == 2) {
            processType2Order(isGift);
        } else {
            System.out.println("Handle other order types");
        }
    }

    private void processType1Order(boolean isPriority, boolean isInternational, boolean isGift, boolean requiresSignature) {
        if (!isPriority) {
            System.out.println("Handle regular domestic order");
            return;
        }
        if (!isInternational) {
            System.out.println("Handle priority domestic order");
            return;
        }
        if (!isGift) {
            System.out.println("Handle international priority order");
            return;
        }
        if (requiresSignature) {
            System.out.println("Handle special case A");
        } else {
            System.out.println("Handle special case B");
        }
    }

    private void processType2Order(boolean isGift) {
        if (isGift) {
            System.out.println("Handle gift order type 2");
        } else {
            System.out.println("Handle regular order type 2");
        }
    }
}
```
